2.2 Add the pricing helpers

Create: client/src/features/screens/pricing/lookup.ts

export type Matrix = {
  meta: { heightMax: number; notes?: string[] };
  table: Record<string, Record<string, number>>;
  housingRules: Array<{ maxW: number; maxH: number; housing: string; roller: string }>;
  adders: {
    remote?: number;
    uChannelPerLf?: number | null;
    colorUpchargePct?: number;
    fabricUpcharges?: Record<string, number>;
  };
  specialQuoteFlags?: { fabrics?: string[] };
};

export function bucket(valueIn: number, stops: number[]) {
  const sorted = [...stops].sort((a, b) => a - b);
  for (const s of sorted) if (valueIn <= s) return s;
  return sorted[sorted.length - 1];
}

export function computeBase(matrix: Matrix, widthIn: number, heightIn: number) {
  const hStops = Object.keys(matrix.table).map(Number);
  const firstRow = Object.values(matrix.table)[0];
  if (!firstRow) throw new Error("matrix table empty");
  const wStops = Object.keys(firstRow).map(Number);
  const h = bucket(heightIn, hStops);
  const w = bucket(widthIn, wStops);
  return matrix.table[String(h)][String(w)];
}

export function selectHousing(matrix: Matrix, w: number, h: number) {
  return (
    matrix.housingRules.find(r => w <= r.maxW && h <= r.maxH) ??
    matrix.housingRules[matrix.housingRules.length - 1]
  );
}

export function addersTotal(matrix: Matrix, opts: {
  remotesQty?: number;
  uChannelLf?: number;
  fabricKey?: string;
  colorNonStandard?: boolean;
}) {
  let extra = 0, pct = 0, notes: string[] = [];
  if (opts.remotesQty && matrix.adders.remote) extra += opts.remotesQty * matrix.adders.remote;
  if (opts.uChannelLf && matrix.adders.uChannelPerLf) extra += opts.uChannelLf * matrix.adders.uChannelPerLf;
  if (opts.colorNonStandard && matrix.adders.colorUpchargePct) pct += matrix.adders.colorUpchargePct;
  if (opts.fabricKey && matrix.adders.fabricUpcharges?.[opts.fabricKey]) pct += matrix.adders.fabricUpcharges[opts.fabricKey];
  if (opts.fabricKey && matrix.specialQuoteFlags?.fabrics?.includes(opts.fabricKey)) notes.push("Special quote required");
  return { extra, pct, notes };
}

export function priceScreen(matrix: Matrix, w: number, h: number, opts: Parameters<typeof addersTotal>[1]) {
  const base = computeBase(matrix, w, h);
  const { extra, pct, notes } = addersTotal(matrix, opts);
  const subtotal = base + extra;
  const total = Math.round(subtotal * (1 + pct));
  const housing = selectHousing(matrix, w, h);
  return { base, extra, pct, total, housing, notes };
}

import React from "react";

export function PriceTile(props: {
  total: number | null;
  housing?: { housing: string; roller: string };
  notes?: string[];
}) {
  return (
    <div className="sticky top-4 rounded-2xl border p-4 shadow-sm">
      <div className="text-xl font-semibold">Estimated Price</div>
      <div className="text-3xl font-bold my-2">
        {props.total !== null ? `$${props.total.toLocaleString()}` : "—"}
      </div>
      {props.housing && (

        </div>
      )}

        <ul className="mt-2 text-xs list-disc ml-4">
          {props.notes.map((n, i) => <li key={i}>{n}</li>)}
        </ul>
      )}
    </div>
  );
}

4.2 The form

Create: client/src/features/screens/components/ScreenForm.tsx

import React from "react";
import { useForm } from "react-hook-form";
import { z } from "zod";
import { zodResolver } from "@hookform/resolvers/zod";
import insectMatrix from "../data/insect_matrix.json";
import { priceScreen, Matrix } from "../../screens/pricing/lookup";

const schema = z.object({
  type: z.enum(["INSECT","SOLAR","VINYL"]).default("INSECT"),
  widthIn: z.coerce.number().min(24).max(240),
  heightIn: z.coerce.number().min(24).max((insectMatrix as any).meta.heightMax || 192),
  remotesQty: z.coerce.number().min(0).max(10).default(1),
  uChannelLf: z.coerce.number().min(0).max(1000).default(0),
  colorNonStandard: z.boolean().default(false),
  fabricKey: z.string().optional()
});

type FormVals = z.infer<typeof schema>;

export default function ScreenForm(props: {
  onAdd: (payload: any) => void;
}) {
  const { register, watch, handleSubmit, formState: { errors } } = useForm<FormVals>({
    resolver: zodResolver(schema),
    defaultValues: { type: "INSECT", widthIn: 72, heightIn: 84, remotesQty: 1, uChannelLf: 0, colorNonStandard: false }
  });

  const vals = watch();
  const matrix = (insectMatrix as unknown) as Matrix;

  const computed = (() => {
    try {
      if (vals.heightIn > matrix.meta.heightMax) return null;
      return priceScreen(matrix, vals.widthIn, vals.heightIn, {
        remotesQty: vals.remotesQty,
        uChannelLf: vals.uChannelLf,
        colorNonStandard: vals.colorNonStandard,
        fabricKey: vals.fabricKey
      });
    } catch {
      return null;
    }
  })();

  const onSubmit = (v: FormVals) => {
    if (!computed) return;
    const { total, housing, notes } = computed;
    props.onAdd({
      sku: `SCREEN-${v.type}-${housing.housing}`,
      name: `${v.type} Screen ${v.widthIn}" x ${v.heightIn}" (${housing.housing}/${housing.roller})`,
      qty: 1,
      unitPrice: total,
      taxable: true,
      meta: { ...v, housing, notes }
    });
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)} className="grid md:grid-cols-3 gap-6">
      <div className="md:col-span-2 space-y-4">
        <div>
          <label className="block text-sm font-medium">Type</label>
          <select {...register("type")} className="mt-1 w-full border rounded-lg p-2">
            <option value="INSECT">Insect (Gaposa)</option>
            <option value="SOLAR">Solar (Gaposa)</option>
            <option value="VINYL">Vinyl Windows (Gaposa)</option>
          </select>
        </div>

        <div className="grid grid-cols-2 gap-4">
          <div>
            <label className="block text-sm font-medium">Width (in)</label>
            <input type="number" {...register("widthIn")} className="mt-1 w-full border rounded-lg p-2" />
            {errors.widthIn && <p className="text-xs text-red-600">{errors.widthIn.message}</p>}
          </div>
          <div>
            <label className="block text-sm font-medium">Height (in)</label>
            <input type="number" {...register("heightIn")} className="mt-1 w-full border rounded-lg p-2" />
            {errors.heightIn && <p className="text-xs text-red-600">{errors.heightIn.message}</p>}
          </div>
        </div>

        <div className="grid grid-cols-3 gap-4">
          <div>
            <label className="block text-sm font-medium">Remotes (qty)</label>
            <input type="number" {...register("remotesQty")} className="mt-1 w-full border rounded-lg p-2" />
          </div>
          <div>
            <label className="block text-sm font-medium">U-Channel (LF)</label>
            <input type="number" {...register("uChannelLf")} className="mt-1 w-full border rounded-lg p-2" />
          </div>
          <div className="flex items-end gap-2">
            <input id="color" type="checkbox" {...register("colorNonStandard")} />
            <label htmlFor="color" className="text-sm">Non-standard color</label>
          </div>
        </div>

        <button type="submit" className="rounded-xl bg-black text-white px-4 py-2">
          Add to Quote
        </button>
      </div>

      <div>
        <div className="mb-4 text-sm text-muted-foreground">
          Max height enforced: <b>{matrix.meta.heightMax}"</b>
        </div>
        <div className="mb-4">
          {computed ? null : <div className="text-sm text-orange-700 bg-orange-50 border border-orange-200 p-2 rounded-lg">Enter valid dimensions to see price</div>}
        </div>
        <div>
          <PriceTile
            total={computed ? computed.total : null}
            housing={computed?.housing}
            notes={computed?.notes}
          />
        </div>
      </div>
    </form>
  );
}
