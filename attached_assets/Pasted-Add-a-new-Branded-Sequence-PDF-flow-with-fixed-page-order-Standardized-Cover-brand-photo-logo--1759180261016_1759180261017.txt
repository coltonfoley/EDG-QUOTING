File: client/src/lib/pdf-branded-sequence.ts (new)

 Implement:

export async function generateBrandedSequencePDF({ pdf, company, quote, renderImages, contractText, showPricing }) {
  const pageW = pdf.internal.pageSize.getWidth();
  const pageH = pdf.internal.pageSize.getHeight();
  const margin = 19;
  const contentW = pageW - margin*2;

  drawStandardCover(pdf, { coverDataUrl: BRAND_COVER_JPG, logoDataUrl: BRAND_LOGO_PNG,
    company, title: quote?.projectName || 'Project Proposal', subtitle: quote?.account?.name || '', pageW, pageH });

  pdf.addPage();
  drawProjectDetailsPage(pdf, { company, quote, margin, contentW, pageW, pageH, showPricing });

  if (renderImages?.length) {
    drawRenderingsPages(pdf, { images: renderImages, margin, contentW, pageW, pageH });
  }

  if ((quote?.items?.length ?? 0) > 0) {
    drawLineItemsSection(pdf, { quote, showPricing, margin, contentW, pageW, pageH });
  }

  if (contractText?.trim()) {
    drawContractSection(pdf, { contractText, margin, contentW, pageW, pageH });
  }

  pdf.addPage();
  drawBrandedBackPage(pdf, { company, pageW, pageH, margin });
}
